cmake_minimum_required(VERSION 3.25)
project(DeckStatus VERSION 2.3.2 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 mit MSVC wird benoetigt. Konfiguration mit -A x64 starten.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)

# One version for code, resources and tests.
configure_file("${CMAKE_SOURCE_DIR}/src/version.h.in" "${CMAKE_BINARY_DIR}/generated/deckstatus_version.h" @ONLY)
include_directories("${CMAKE_BINARY_DIR}/generated")

add_library(DeckStatusBridge SHARED src/deckstatus_bridge.cpp src/rekordbox_profile.h src/deckstatus_protocol.h src/deckstatus.rc)
target_link_libraries(DeckStatusBridge PRIVATE version bcrypt)
target_compile_definitions(DeckStatusBridge PRIVATE DECKSTATUS_BRIDGE)

add_library(deckstatus_audio STATIC src/audio_capture.cpp src/audio_capture.h src/audio_samples.h)
target_include_directories(deckstatus_audio PUBLIC vendor)
target_link_libraries(deckstatus_audio PUBLIC ole32 uuid)
add_library(deckstatus_prolink STATIC src/prolink.cpp src/prolink.h)
target_include_directories(deckstatus_prolink PUBLIC vendor)
target_link_libraries(deckstatus_prolink PUBLIC crypt32)
add_library(deckstatus_network STATIC src/network.cpp src/network.h)
target_include_directories(deckstatus_network PUBLIC vendor)
target_link_libraries(deckstatus_network PUBLIC iphlpapi ws2_32)
add_library(deckstatus_portal STATIC src/portal.cpp src/portal.h src/portal_http.h src/master_gate.h src/twitch.cpp src/twitch.h src/automation.h src/updater.cpp src/updater.h)
target_include_directories(deckstatus_portal PUBLIC vendor)
target_link_libraries(deckstatus_portal PUBLIC bcrypt crypt32 winhttp)
add_executable(DeckStatus src/deckstatus.cpp src/injector.cpp src/server.cpp src/artwork.cpp src/deckstatus.rc assets/DeckStatus.ico)
target_include_directories(DeckStatus PRIVATE vendor vendor/httplib assets)
target_link_libraries(DeckStatus PRIVATE version ws2_32 crypt32 deckstatus_audio deckstatus_prolink deckstatus_network deckstatus_portal)
add_custom_target(DeckStatusWeb
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:DeckStatus>/web"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/tools/DeckStatus.Update.ps1" "$<TARGET_FILE_DIR:DeckStatus>/DeckStatus.Update.ps1")
add_dependencies(DeckStatus DeckStatusBridge DeckStatusWeb)

install(TARGETS DeckStatus DeckStatusBridge RUNTIME DESTINATION .)
install(DIRECTORY web DESTINATION .)
install(FILES README.md CHANGELOG.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(DIRECTORY "${CMAKE_BINARY_DIR}/prolink-package/" DESTINATION prolink)
install(FILES Start-ProLink.cmd DESTINATION .)
install(FILES tools/DeckStatus.Update.ps1 DESTINATION .)
install(FILES vendor/httplib/LICENSE DESTINATION vendor/httplib)
install(FILES vendor/nlohmann/LICENSE.MIT DESTINATION vendor/nlohmann)
